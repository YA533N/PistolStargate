extends Node
var _music := AudioStreamPlayer.new()

func _ready() -> void:
	_music.volume_db = -20.0
	add_child(_music)

## Plays a song. If that song is already playing, it keeps going instead of restarting.
func play_music(stream: AudioStream) -> void:
	if _music.stream == stream and _music.playing:
		return
	_music.stream = stream
	_music.play()

	add_child(p)
	p.play()

class_name Chicken
extends Enemy
## The Proving Grounds' first "enemy". Completely harmless: it wanders, pecks at the floor, clucks,
## and runs from the player. The model has no animations, so the waddle and pecking are done in code.

const CLUCKS := ["BAWK!", "bok?", "BAWK BAWK!", "bok bok", "...bawk", "BAAAWK!"]

@export var walk_speed := 1.4
@export var panic_speed := 2.6
@export var panic_distance := 3.5
## How far from its spawn point it wanders.
@export var wander_radius := 3.0

@onready var _model: Node3D = $Model
@onready var _cluck: Label3D = $Cluck

var _home := Vector3.ZERO
var _wander_target := Vector3.ZERO
var _peck_time := 0.0
var _anim_time := 0.0
var _cluck_time := 1.5


func _ready() -> void:
	super()
	_home = global_position
	_wander_target = _home
	_cluck.visible = false


func _physics_process(delta: float) -> void:
	apply_gravity(delta)
	var target_velocity := Vector3.ZERO
	var player := get_player()
	var panicking := false
	if player and player.global_position.distance_to(global_position) < panic_distance:
		var away := global_position - player.global_position
		# Don't run off forever: lean back toward home when far from it.
		away += (_home - global_position) * 0.25
		away.y = 0.0
		target_velocity = away.normalized() * panic_speed
		panicking = true
		_peck_time = 0.0
	elif _peck_time > 0.0:
		_peck_time -= delta
	else:
		var to_target := _wander_target - global_position
		to_target.y = 0.0
		if to_target.length() < 0.4:
			_peck_time = randf_range(0.8, 2.0)
			_pick_wander_target()
		else:
			target_velocity = to_target.normalized() * walk_speed
	velocity.x = move_toward(velocity.x, target_velocity.x, 20.0 * delta)
	velocity.z = move_toward(velocity.z, target_velocity.z, 20.0 * delta)
	move_and_slide()
	if is_on_wall():
		_pick_wander_target()
	face(velocity, delta, 12.0)
	_animate(delta, Vector2(velocity.x, velocity.z).length())
	_tick_cluck(delta, panicking)


func _pick_wander_target() -> void:
	_wander_target = _home + Vector3(randf_range(-wander_radius, wander_radius), 0.0, randf_range(-wander_radius, wander_radius))


func _animate(delta: float, speed: float) -> void:
	_anim_time += delta * (5.0 + speed * 5.0)
	var moving := speed > 0.2
	var hop := absf(sin(_anim_time)) * 0.08 if moving else 0.0
	var waddle := sin(_anim_time) * 0.14 if moving else 0.0
	var peck := maxf(0.0, sin(_anim_time * 1.6)) * 0.55 if _peck_time > 0.0 else 0.0
	_model.position.y = hop
	_model.rotation.z = lerpf(_model.rotation.z, waddle, 0.3)
	_model.rotation.x = lerpf(_model.rotation.x, peck, 0.3)


func _tick_cluck(delta: float, panicking: bool) -> void:
	_cluck_time -= delta * (3.0 if panicking else 1.0)
	if _cluck_time > 0.0:
		return
	_cluck_time = randf_range(2.0, 4.0)
	_cluck.text = CLUCKS.pick_random()
	_cluck.visible = true
	_cluck.modulate.a = 1.0
	var tween := create_tween()
	tween.tween_interval(0.8)
	tween.tween_property(_cluck, "modulate:a", 0.0, 0.3)
	tween.tween_callback(_cluck.hide)


func _die() -> void:
	Effects.flash(self, global_position + Vector3.UP * 0.5, Color(1, 1, 1), 1.4, 0.35)
	Effects.float_text(self, global_position + Vector3.UP * 1.4, "*bawk...*", Color(1, 0.9, 0.7), 0.9)
	queue_free()

class_name Enemy
extends CharacterBody3D
## Base for everything the player can kill: health, gravity, loot and kill credit.
## tier decides what the Infinite Pistol does: NORMAL enemies die in one shot and spill a jackpot;
## ELITE and BOSS enemies only take its elite_damage.
## Expects a Health child. Put enemies on the "enemies" physics layer (3).

signal killed(enemy: Enemy)

enum Tier { NORMAL, ELITE, BOSS }

@export var display_name := "Enemy"
@export var tier := Tier.NORMAL
@export var gravity := 24.0

@export_group("Loot")
## Cash dropped on a normal kill, from x to y.
@export var money_drop := Vector2i(3, 8)
@export_range(0.0, 1.0) var ammo_drop_chance := 0.35
@export_range(0.0, 1.0) var health_drop_chance := 0.15
@export_range(0.0, 1.0) var weapon_drop_chance := 0.1
## Always part of the Infinite Pistol jackpot (the tutorial chicken uses this). Random weapons fill the rest.
@export var jackpot_weapons: Array[WeaponData] = []

@onready var health: Health = $Health

## Set by Combat when the Infinite Pistol lands the killing shot.
var killed_by_infinite_pistol := false


func _ready() -> void:
	add_to_group("enemies")
	health.show_damage_numbers = true
	health.died.connect(_on_died)


func is_dead() -> bool:
	return health.is_dead()


func get_player() -> Player:
	return get_tree().get_first_node_in_group("player") as Player


func apply_gravity(delta: float) -> void:
	if is_on_floor():
		velocity.y = minf(velocity.y, 0.0)
	else:
		velocity.y -= gravity * delta


## Turns smoothly toward direction. Imported characters face +Z, so the yaw is atan2(x, z).
func face(direction: Vector3, delta: float, sharpness := 10.0) -> void:
	direction.y = 0.0
	if direction.length_squared() < 0.0001:
		return
	rotation.y = lerp_angle(rotation.y, atan2(direction.x, direction.z), 1.0 - exp(-sharpness * delta))


func _on_died() -> void:
	remove_from_group("enemies")
	set_deferred("collision_layer", 0)
	GameState.enemy_killed.emit(self)
	if killed_by_infinite_pistol:
		Loot.jackpot(self)
	else:
		Loot.normal_drop(self)
	killed.emit(self)
	_die()


## Default death: a puff and gone. Bigger enemies override this to play an animation.
func _die() -> void:
	Effects.flash(self, global_position + Vector3.UP * 0.6, Color(1, 0.95, 0.8), 1.2, 0.3)
	queue_free()

class_name FirstDescendant
extends Enemy
## Boss of the Proving Grounds: a heavy with a machine gun. It drops in from the sky, walks to mid range,
## then alternates two attacks, each telegraphed by a short wind-up (a red "!" and a flashing light):
##   SWEEP - a spray that sweeps an arc across the player, with gaps to slip through.
##   BURST - three quick aimed bursts at where the player is standing (punishes standing still).
## Bullets leave the gun and dip to knee height, below the top of the low walls: walls and crates are cover,
## and a dash (invincible) or a well-timed jump gets you through.
## Drops ammo at 75% and 50% health and health at 25%, so the fight can't stall when the player runs dry.

enum State { FALLING, LANDED, WALK, WINDUP, SWEEP, BURST, REST, DEAD }

const ANIM_DEATH := "_Action Stash]"
const ANIM_WALK := "_Action Stash]_001"
const ANIM_FIRE := "_Action Stash]_002"
const MUZZLE_BONE := "FX_Muzzle"
## Bullets fly this high above the boss's feet: under the top of the 0.8 m low walls, so they block them.
const BULLET_HEIGHT := 0.5

@export var walk_speed := 2.6
## Distance it tries to keep from the player.
@export var preferred_range := 9.0

@export_group("Gun")
@export var bullet_damage := 6
@export var bullet_speed := 15.0
@export var sweep_rate := 12.0
@export var sweep_duration := 2.0
@export var sweep_arc_degrees := 70.0
@export var burst_count := 3
@export var burst_size := 4
@export var windup_time := 0.7
@export var rest_time := 1.4

@onready var _model: Node3D = $Model
@onready var _warning: Label3D = $Warning
@onready var _warning_light: OmniLight3D = $WarningLight

var state := State.FALLING

var _state_time := 0.0
var _walk_time := 0.0
var _anim: AnimationPlayer
var _muzzle: Node3D
var _gun := WeaponData.new()
var _aim_yaw := 0.0
var _sweep_direction := 1.0
var _shot_timer := 0.0
var _shots_fired := 0
var _next_attack_is_sweep := true
## Health fractions where it drops supplies.
var _drops_left := [0.75, 0.5, 0.25]


func _ready() -> void:
	super()
	health.invulnerable = true
	health.changed.connect(_on_health_changed)
	_gun.damage = bullet_damage
	_gun.projectile_speed = bullet_speed
	_gun.projectile_size = 0.2
	_gun.projectile_color = Color(1, 0.3, 0.15)
	_gun.projectile_range = 40.0
	_anim = _model.find_children("*", "AnimationPlayer", true, false)[0]
	for animation in [ANIM_WALK, ANIM_FIRE]:
		_anim.get_animation(animation).loop_mode = Animation.LOOP_LINEAR
	_anim.play(ANIM_WALK)
	_attach_muzzle()
	_show_warning(false)


func _physics_process(delta: float) -> void:
	_state_time += delta
	apply_gravity(delta)
	var player := get_player()
	match state:
		State.FALLING:
			if is_on_floor():
				_land()
		State.LANDED:
			_set_horizontal(Vector3.ZERO, delta)
			if _state_time > 1.2:
				_enter(State.WALK)
		State.WALK:
			_walk(delta, player)
		State.WINDUP:
			_windup(delta, player)
		State.SWEEP:
			_sweep(delta)
		State.BURST:
			_burst(delta, player)
		State.REST:
			_walk(delta, player, 0.5)
			if _state_time > rest_time:
				_enter(State.WALK)
	move_and_slide()


func _enter(new_state: State) -> void:
	state = new_state
	_state_time = 0.0
	_show_warning(new_state == State.WINDUP)
	match new_state:
		State.WALK:
			_anim.play(ANIM_WALK)
			_walk_time = randf_range(1.2, 2.2)
		State.WINDUP:
			_anim.play(ANIM_FIRE)
			_anim.pause()
		State.SWEEP:
			_anim.play(ANIM_FIRE)
			_shot_timer = 0.0
			_shots_fired = 0
			_aim_yaw = _yaw_to_player() - deg_to_rad(sweep_arc_degrees) * 0.5 * _sweep_direction
		State.BURST:
			_anim.play(ANIM_FIRE)
			_shots_fired = 0
		State.REST:
			_anim.play(ANIM_WALK)


func _land() -> void:
	health.invulnerable = false
	Effects.flash(self, global_position + Vector3.UP * 0.3, Color(1, 0.4, 0.2), 5.0, 0.5)
	Effects.shake(self, 0.8)
	var player := get_player()
	if player and player.global_position.distance_to(global_position) < 3.5:
		player.health.take_damage(15)
	get_tree().call_group("hud", "track_boss", self)
	_enter(State.LANDED)


## Walks to preferred range, strafing once there. speed_scale slows it down while resting.
func _walk(delta: float, player: Player, speed_scale := 1.0) -> void:
	if player == null:
		_set_horizontal(Vector3.ZERO, delta)
		return
	var to_player := player.global_position - global_position
	to_player.y = 0.0
	var distance := to_player.length()
	var move := Vector3.ZERO
	if distance > preferred_range + 1.5:
		move = to_player.normalized()
	elif distance < preferred_range - 3.0:
		move = -to_player.normalized()
	else:
		move = to_player.normalized().cross(Vector3.UP) * _sweep_direction * 0.6
	_set_horizontal(move * walk_speed * speed_scale, delta)
	face(to_player, delta, 6.0)
	if state == State.WALK and _state_time > _walk_time:
		_enter(State.WINDUP)


func _windup(delta: float, player: Player) -> void:
	_set_horizontal(Vector3.ZERO, delta)
	if player:
		face(player.global_position - global_position, delta, 8.0)
	_warning_light.light_energy = 3.0 + sin(_state_time * 30.0) * 2.0
	if _state_time >= windup_time:
		_enter(State.SWEEP if _next_attack_is_sweep else State.BURST)
		_next_attack_is_sweep = not _next_attack_is_sweep


func _sweep(delta: float) -> void:
	_set_horizontal(Vector3.ZERO, delta)
	_aim_yaw += _sweep_direction * deg_to_rad(sweep_arc_degrees) / sweep_duration * delta
	rotation.y = _aim_yaw
	_shot_timer -= delta
	while _shot_timer <= 0.0:
		_shot_timer += 1.0 / sweep_rate
		_shots_fired += 1
		# Every 4th bullet is skipped, leaving gaps in the wall of fire.
		if _shots_fired % 4 != 0:
			_shoot(_aim_yaw + randf_range(-0.02, 0.02))
	if _state_time >= sweep_duration:
		_sweep_direction = -_sweep_direction
		_enter(State.REST)


func _burst(delta: float, player: Player) -> void:
	_set_horizontal(Vector3.ZERO, delta)
	if player:
		face(player.global_position - global_position, delta, 10.0)
	var total := burst_count * burst_size
	while _shots_fired < total and _state_time >= _burst_shot_time(_shots_fired):
		_shoot(_yaw_to_player() + randf_range(-0.06, 0.06))
		_shots_fired += 1
	if _shots_fired >= total and _state_time > _burst_shot_time(total - 1) + 0.3:
		_enter(State.REST)


## When the nth bullet of a burst attack is due: bursts 0.45 s apart, bullets 0.07 s apart.
func _burst_shot_time(n: int) -> float:
	return floori(float(n) / burst_size) * 0.45 + (n % burst_size) * 0.07


func _shoot(yaw: float) -> void:
	var direction := Vector3(sin(yaw), 0.0, cos(yaw))
	var bullet := Projectile.new()
	bullet.cruise_height = global_position.y + BULLET_HEIGHT
	get_tree().current_scene.add_child(bullet)
	bullet.launch(_gun, _muzzle.global_position, direction, Combat.ENEMY_SHOT_MASK)
	Effects.flash(self, _muzzle.global_position, _gun.projectile_color, 0.35, 0.06)


## Measured from the muzzle, not the body: the gun is held about 0.6 m to the side, so aiming from the
## body's centre sent every bullet past the player's shoulder. A player closer than the muzzle (it sticks
## out 2.5 m) is aimed at from the body instead, or the shots would turn backwards.
func _yaw_to_player() -> float:
	var player := get_player()
	if player == null:
		return rotation.y
	var from := _muzzle.global_position
	if global_position.distance_to(player.global_position) < global_position.distance_to(from) + 1.0:
		from = global_position
	var offset := player.global_position - from
	return atan2(offset.x, offset.z)


func _set_horizontal(target: Vector3, delta: float) -> void:
	velocity.x = move_toward(velocity.x, target.x, 20.0 * delta)
	velocity.z = move_toward(velocity.z, target.z, 20.0 * delta)


func _show_warning(on: bool) -> void:
	_warning.visible = on
	_warning_light.visible = on


func _attach_muzzle() -> void:
	var skeletons := _model.find_children("*", "Skeleton3D", true, false)
	var skeleton := skeletons[0] as Skeleton3D if not skeletons.is_empty() else null
	if skeleton and skeleton.find_bone(MUZZLE_BONE) >= 0:
		var attachment := BoneAttachment3D.new()
		attachment.bone_name = MUZZLE_BONE
		skeleton.add_child(attachment)
		_muzzle = attachment
	else:
		push_warning("First Descendant model has no %s bone; firing from its chest." % MUZZLE_BONE)
		_muzzle = Marker3D.new()
		_muzzle.position = Vector3(0, 1.4, 1.2)
		add_child(_muzzle)


func _on_health_changed(current: int, maximum: int) -> void:
	while not _drops_left.is_empty() and current > 0 and current <= maximum * _drops_left[0]:
		var threshold: float = _drops_left.pop_front()
		var kind := LootPickup.Kind.HEALTH if threshold <= 0.25 else LootPickup.Kind.AMMO
		for i in 2:
			Loot.spawn_item(self, kind, 25, global_position + Vector3.UP * 2.0, Loot.landing_spot(self, global_position, 2.5, 4.0))
		Effects.float_text(self, global_position + Vector3.UP * 4.4, "ARMOR CRACKED!", Color(1, 0.6, 0.2), 1.1)


func _die() -> void:

	set_physics_process(false)

	var tween := create_tween()
	tween.tween_interval(3.0)
	tween.tween_property(self, "position:y", position.y - 3.5, 2.0)
	tween.tween_callback(queue_free)

class_name SpikeTrap
extends Area3D
## Hidden floor spikes that teach "keep moving". Step on one and it clicks and glows red; a moment later the
## spikes shoot up and hurt whoever is still standing there. With cycle_time set it also fires on its own,
## so a field of traps with different offsets makes a moving pattern.
## The imported animation retracts the spikes, so extending them plays it backwards.

enum State { HIDDEN, ARMED, UP, COOLDOWN }

@export var damage := 15
## Seconds between being triggered and the spikes coming up.
@export var arm_time := 0.55
@export var up_time := 1.0
@export var cooldown := 0.6
## Above 0, the trap also fires by itself every cycle_time seconds, starting after cycle_offset.
@export var cycle_time := 0.0
@export var cycle_offset := 0.0
## Upward pop given to the player when the spikes hit them.
@export var launch_speed := 6.0

@onready var _model: Node3D = $Model
@onready var _glow: OmniLight3D = $Glow
@onready var _warning: MeshInstance3D = $Warning

var state := State.HIDDEN

var _anim: AnimationPlayer
var _animation := ""
var _time := 0.0
var _cycle := 0.0
var _bodies: Array[Node3D] = []
var _already_hit: Array[Node3D] = []
var _warning_material := StandardMaterial3D.new()


func _ready() -> void:
	_anim = _model.find_children("*", "AnimationPlayer", true, false)[0]
	_animation = _anim.get_animation_list()[0]
	# Start with the spikes hidden: the last frame of the retract animation.
	_anim.play(_animation)
	_anim.seek(_anim.get_animation(_animation).length, true)
	_anim.pause()
	_warning_material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
	_warning_material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
	_warning_material.albedo_color = Color(1, 0.1, 0.05, 0.0)
	_warning.material_override = _warning_material
	_glow.light_energy = 0.0
	_cycle = -cycle_offset
	body_entered.connect(_on_body_entered)
	body_exited.connect(_on_body_exited)


func _physics_process(delta: float) -> void:
	_time += delta
	match state:
		State.HIDDEN:
			if cycle_time > 0.0:
				_cycle += delta
				if _cycle >= cycle_time:
					_cycle = 0.0
					_arm()
		State.ARMED:
			var flash := 0.5 + 0.5 * sin(_time * 28.0)
			_warning_material.albedo_color.a = 0.25 + 0.35 * flash
			_glow.light_energy = 2.5 * flash
			if _time >= arm_time:
				_extend()
		State.UP:
			_hurt_bodies()
			if _time >= up_time:
				_retract()
		State.COOLDOWN:
			if _time >= cooldown:
				state = State.HIDDEN
				if not _bodies.is_empty():
					_arm()


func _arm() -> void:
	state = State.ARMED
	_time = 0.0


func _extend() -> void:
	state = State.UP
	_time = 0.0
	_already_hit.clear()
	_anim.play(_animation, -1, -2.5, true)
	_warning_material.albedo_color.a = 0.15
	_glow.light_energy = 1.0


func _retract() -> void:
	state = State.COOLDOWN
	_time = 0.0
	_anim.play(_animation)
	_warning_material.albedo_color.a = 0.0
	_glow.light_energy = 0.0


func _hurt_bodies() -> void:
	# Give the spikes a moment to actually come up before they count.
	if _time < 0.08:
		return
	for body in _bodies:
		if body in _already_hit:
			continue
		_already_hit.append(body)
		var health := Combat.health_of(body)
		if health:
			health.take_damage(damage)
		var player := body as Player
		if player and not player.health.invulnerable:
			player.velocity.y = launch_speed


func _on_body_entered(body: Node3D) -> void:
	if not body in _bodies:
		_bodies.append(body)
	if state == State.HIDDEN:
		_arm()


func _on_body_exited(body: Node3D) -> void:
	_bodies.erase(body)

class_name NPC
extends Node3D
## A character the player can talk to. When the player is inside TalkArea a "[E] Talk" prompt shows;
## pressing interact opens the DialogueBox with this NPC's lines and freezes the player until it closes.
## Level scripts can also drive a conversation with say() and ask() (the Proving Grounds does).
## Expects children: Model (an imported character), TalkArea (Area3D) and Prompt (Label3D).

## Emitted after a player-started conversation closes.
signal conversation_finished

@export var speaker_name := "???"
@export var lines := PackedStringArray()
## Used instead of lines once the Proving Grounds is done. Leave empty to always use lines.
@export var lines_after_tutorial := PackedStringArray()
## Off: no prompt and E does nothing (for NPCs that only talk in scripted scenes).
@export var can_talk := true
## Sends the player to this scene after the conversation, with a teleport flash. Empty: stay here.
@export_file("*.tscn") var send_to_scene := ""
## Animation to loop on the model. Leave empty to use the model's first animation.
@export var idle_animation := ""
## Hold the animation's first frame and only move while talking. Off: loop it all the time.
@export var animate_only_while_talking := true
## Turn toward the player when a conversation starts.
@export var face_player := true

@onready var _model: Node3D = $Model
@onready var _talk_area: Area3D = $TalkArea
@onready var _prompt: Label3D = $Prompt

## The player inside the talk area, or null.
var _player_in_range: Player
## The player currently in a conversation with this NPC, or null.
var _talking_to: Player
var _animation_player: AnimationPlayer


func _ready() -> void:
	_prompt.visible = false
	_talk_area.body_entered.connect(_on_body_entered)
	_talk_area.body_exited.connect(_on_body_exited)
	_setup_animation()


func _unhandled_input(event: InputEvent) -> void:
	if not can_talk or _player_in_range == null or _talking_to != null or not event.is_action_pressed("talk"):
		return
	var box := _dialogue_box()
	if box == null or box.is_open():
		return
	get_viewport().set_input_as_handled()
	_start_talking(box)


## Scripted line(s) for cutscenes: animates, faces the player and returns once the box is closed.
func say(text: PackedStringArray) -> void:
	var box := _dialogue_box()
	if box == null:
		return
	_begin_scripted()
	box.start(speaker_name, text)
	await box.finished
	_end_scripted()


## Scripted question. Returns the index of the answer the player picked.
func ask(question: String, options: PackedStringArray) -> int:
	var box := _dialogue_box()
	if box == null:
		return 0
	_begin_scripted()
	box.ask(speaker_name, question, options)
	var answer: int = await box.choice_made
	_end_scripted()
	return answer


func _start_talking(box: DialogueBox) -> void:
	_talking_to = _player_in_range
	_talking_to.input_enabled = false
	_prompt.visible = false
	if face_player:
		_turn_toward(_talking_to.global_position)
	if animate_only_while_talking and _animation_player:
		_animation_player.play()
	box.finished.connect(_on_dialogue_finished, CONNECT_ONE_SHOT)
	var text := lines_after_tutorial if GameState.proving_grounds_done and not lines_after_tutorial.is_empty() else lines
	box.start(speaker_name, text)


func _on_dialogue_finished() -> void:
	var player := _talking_to
	_talking_to = null
	if animate_only_while_talking and _animation_player:
		_hold_first_frame()
	conversation_finished.emit()
	if not send_to_scene.is_empty():
		# The player stays frozen through the teleport.
		Effects.flash(self, player.global_position + Vector3.UP, Color(1, 0.6, 0.2), 2.5, 0.6)
		Transition.change_scene(send_to_scene, Color(1, 0.75, 0.4))
		return
	player.input_enabled = true
	_prompt.visible = _player_in_range != null and can_talk


func _begin_scripted() -> void:
	var player := get_tree().get_first_node_in_group("player") as Player
	if face_player and player:
		_turn_toward(player.global_position)
	if _animation_player:
		_animation_player.play()


func _end_scripted() -> void:
	if animate_only_while_talking and _animation_player:
		_hold_first_frame()


func _dialogue_box() -> DialogueBox:

	_time += delta

		_hidden_time -= delta

			position = _home

class_name Player
extends CharacterBody3D


signal dash_charges_changed(charges: int, max_charges: int)

@export_group("Movement")
## Metres per second. Placeholder.
@export var move_speed := 8.0
@export var ground_acceleration := 70.0
@export var air_acceleration := 35.0

@export_group("Jump")
## Placeholder. With gravity 24 this gives a jump height of about 1.3 m.
@export var jump_velocity := 8.0
@export var gravity := 24.0
@export var max_fall_speed := 30.0

@export_group("Dash")
## GDD: 2 grid units, any direction, on the ground or in the air.
@export var dash_distance_grid := 2.0
@export var dash_duration := 0.12
## Placeholder.
@export var dash_cooldown := 0.6
## GDD: one charge.
@export var max_dash_charges := 1
## Placeholder: invincibility frames for the whole dash.
@export var dash_invulnerable := true

@export_group("Slide")
## Starting speed of a slide. It fades out over the slide. Placeholder.
@export var slide_speed := 14.0
## Placeholder.
@export var slide_duration := 0.5

@export_group("Grapple")
## Max distance to a grapple point. Placeholder.
@export var grapple_range := 20.0
## Sideways push from WASD while swinging. Placeholder.
@export var swing_acceleration := 30.0
## How fast the rope shortens while held, in metres per second. Placeholder.
@export var reel_speed := 3.0
## The rope never reels in shorter than this. Placeholder.
@export var min_rope_length := 1.5
## How strongly a stretched rope pulls you back in. Placeholder.
@export var rope_stiffness := 10.0
## Fraction of speed lost per second while swinging. Placeholder.
@export var rope_drag := 0.3

## Falling below this height respawns the player at their start point.
@export var kill_height := -20.0

@onready var health: Health = $Health
@onready var body: Node3D = $Body
@onready var arsenal: Arsenal = $Arsenal

## Where the mouse cursor points, at the held gun's muzzle height. Weapons fire toward it.
var aim_point := Vector3.ZERO
var dash_charges := 0
## Turned off during dialogue. The player still falls and finishes a dash, but ignores move/jump/dash/shoot.
var input_enabled := true

var _dash_time_left := 0.0
var _dash_direction := Vector3.ZERO
var _recharge_time_left := 0.0
var _spawn_position := Vector3.ZERO
var _slide_time_left := 0.0
var _slide_direction := Vector3.ZERO
## The grapple_point node we are hanging from. Null when not grappling.
var _grapple_target: Node3D = null
var _rope_length := 0.0


func _ready() -> void:
	dash_charges = max_dash_charges
	_spawn_position = global_position
	aim_point = global_position - global_basis.z
	health.died.connect(_on_died)


func _physics_process(delta: float) -> void:
	_update_aim()
	_tick_dash_recharge(delta)

	if input_enabled and not is_dashing() and dash_charges > 0 and Input.is_action_just_pressed("dash"):
		_release_grapple()  # dashing cancels the grapple
		_end_slide()  # and the slide
		_start_dash()

	if input_enabled and not is_dashing() and not is_grappling() and not is_sliding() and Input.is_action_just_pressed("slide"):
		_start_slide()

	if input_enabled and not is_dashing() and not is_grappling() and Input.is_action_just_pressed("grapple"):
		_start_grapple()
		if is_grappling():
			_end_slide()

	var was_dashing := is_dashing()
	if was_dashing:
		_process_dash(delta)
	elif is_grappling():
		_process_grapple(delta)
	elif is_sliding():
		_process_slide(delta)
	else:
		_process_movement(delta)

	move_and_slide()

	# Ended after moving, so the dash's last partial step still uses dash speed.
	if was_dashing and not is_dashing():
		_end_dash()

	arsenal.update(aim_point, input_enabled, delta)

	if global_position.y < kill_height:
		_respawn()


func is_dashing() -> bool:
	return _dash_time_left > 0.0


func is_grappling() -> bool:
	return _grapple_target != null


func is_sliding() -> bool:
	return _slide_time_left > 0.0


## 0 right after a dash, 1 when the next charge is ready. For the HUD.
func dash_recharge_progress() -> float:
	if dash_charges >= max_dash_charges or dash_cooldown <= 0.0:
		return 1.0
	return 1.0 - _recharge_time_left / dash_cooldown


func _process_movement(delta: float) -> void:
	var input := Vector2.ZERO
	if input_enabled:
		input = Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	var target_velocity := _camera_relative(input) * move_speed
	var acceleration := ground_acceleration if is_on_floor() else air_acceleration
	var horizontal := Vector3(velocity.x, 0.0, velocity.z).move_toward(target_velocity, acceleration * delta)
	velocity.x = horizontal.x
	velocity.z = horizontal.z

	if is_on_floor():
		if input_enabled and Input.is_action_just_pressed("jump"):
			velocity.y = jump_velocity
	else:
		velocity.y = maxf(velocity.y - gravity * delta, -max_fall_speed)


func _start_dash() -> void:
	var input := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	var direction := _camera_relative(input)
	if direction == Vector3.ZERO:
		# No movement key held: dash toward where the player is aiming.
		direction = -body.global_basis.z
		direction.y = 0.0
	_dash_direction = direction.normalized()
	_dash_time_left = dash_duration

	if dash_charges == max_dash_charges:
		_recharge_time_left = dash_cooldown
	dash_charges -= 1
	health.invulnerable = dash_invulnerable
	dash_charges_changed.emit(dash_charges, max_dash_charges)


func _process_dash(delta: float) -> void:
	# Scale the last frame's step so the dash covers exactly the set distance.
	var step := minf(delta, _dash_time_left)
	var speed := Grid.to_world(dash_distance_grid) / dash_duration
	# y stays 0, so a dash ignores gravity, including in the air.
	velocity = _dash_direction * (speed * step / delta)
	_dash_time_left -= step


## Needs a movement key held, so a slide always has a direction.
func _start_slide() -> void:
	if not is_on_floor():
		return
	var input := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	var direction := _camera_relative(input)
	if direction == Vector3.ZERO:
		return
	_slide_direction = direction.normalized()
	_slide_time_left = slide_duration


## Burst of speed that fades out. Jumping, running off a ledge or running out of time ends it.
func _process_slide(delta: float) -> void:
	_slide_time_left -= delta
	if _slide_time_left <= 0.0 or not is_on_floor() or not input_enabled or Input.is_action_just_pressed("jump"):
		_end_slide()
		# Hands over to normal movement this same frame, which also handles the jump.
		_process_movement(delta)
		return

	var speed := slide_speed * (0.4 + 0.6 * _slide_time_left / slide_duration)
	velocity.x = _slide_direction.x * speed
	velocity.z = _slide_direction.z * speed
	# Keeps the player pressed to the floor on slopes.
	velocity.y = maxf(velocity.y - gravity * delta, -max_fall_speed)


func _end_slide() -> void:
	_slide_time_left = 0.0


## Picks the grapple_point nearest the cursor (measured flat on the ground plane) within range.
func _start_grapple() -> void:
	var best: Node3D = null
	var best_dist := INF
	for node in get_tree().get_nodes_in_group("grapple_point"):
		var point := node as Node3D
		if point == null or point.global_position.distance_to(global_position) > grapple_range:
			continue
		var flat := point.global_position
		flat.y = aim_point.y
		var d := flat.distance_to(aim_point)
		if d < best_dist:
			best_dist = d
			best = point
	_grapple_target = best
	if _grapple_target:
		_rope_length = _grapple_target.global_position.distance_to(global_position)


## Rope swing: gravity and WASD push you around, and the rope only stops you moving away from the anchor.
func _process_grapple(delta: float) -> void:
	if not is_instance_valid(_grapple_target) or not input_enabled \
			or Input.is_action_just_released("grapple") or Input.is_action_just_pressed("jump"):
		_release_grapple()
		return

	_rope_length = maxf(_rope_length - reel_speed * delta, min_rope_length)

	var input := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	velocity.y = maxf(velocity.y - gravity * delta, -max_fall_speed)
	velocity += _camera_relative(input) * swing_acceleration * delta
	velocity *= 1.0 - rope_drag * delta

	var to_anchor := _grapple_target.global_position - global_position
	var dist := to_anchor.length()
	if dist > _rope_length and dist > 0.001:
		var rope_dir := to_anchor / dist
		# Cancel only the part of the velocity that points away from the anchor.
		var away := velocity.dot(-rope_dir)
		if away > 0.0:
			velocity += rope_dir * away
		velocity += rope_dir * (dist - _rope_length) * rope_stiffness


## Velocity is kept, so letting go flings you along your swing.
func _release_grapple() -> void:
	_grapple_target = null


func _end_dash() -> void:
	_dash_time_left = 0.0
	# Keep some momentum instead of stopping dead.
	velocity = _dash_direction * move_speed
	health.invulnerable = false


func _tick_dash_recharge(delta: float) -> void:
	if dash_charges >= max_dash_charges:
		return
	_recharge_time_left -= delta
	if _recharge_time_left <= 0.0:
		dash_charges += 1
		_recharge_time_left = dash_cooldown if dash_charges < max_dash_charges else 0.0
		dash_charges_changed.emit(dash_charges, max_dash_charges)


## Turns WASD input into a world direction, so "up" on the keyboard is "up" on screen.
func _camera_relative(input: Vector2) -> Vector3:
	var camera := get_viewport().get_camera_3d()
	if camera == null:
		return Vector3(input.x, 0.0, input.y)
	var forward := -camera.global_basis.z
	forward.y = 0.0
	var right := camera.global_basis.x
	right.y = 0.0
	return (right.normalized() * input.x - forward.normalized() * input.y).limit_length(1.0)


func _update_aim() -> void:
	var camera := get_viewport().get_camera_3d()
	if camera == null:
		return
	var mouse := get_viewport().get_mouse_position()
	var ray_origin := camera.project_ray_origin(mouse)
	var ray_normal := camera.project_ray_normal(mouse)
	# Aim at gun height, not the floor: with the tilted camera, a bullet passing over a floor point shows up
	# above it on screen, so it would miss the cursor.
	var muzzle_height := arsenal.held.muzzle_position().y if arsenal.held else global_position.y + 1.0
	var aim_height := muzzle_height
	# What's under the cursor: an enemy, or the floor / a prop.
	var under := get_world_3d().direct_space_state.intersect_ray(
			PhysicsRayQueryParameters3D.create(ray_origin, ray_origin + ray_normal * 500.0, Combat.PLAYER_SHOT_MASK))
	var hit: Variant = null
	if not under.is_empty() and not is_on_floor() and Combat.health_of(under.collider):
		# In the air (grapple, jump) with the cursor on an enemy: shoot down at exactly that spot on it.
		hit = under.position
	else:
		# Up high, aim at gun height above the ground under the cursor, so shots angle down to enemies below.
		# Never higher than the gun: on flat ground this is the same height and shots stay flat.
		if not under.is_empty():
			aim_height = minf(muzzle_height, under.position.y + muzzle_height - global_position.y)
		hit = Plane(Vector3.UP, aim_height).intersects_ray(ray_origin, ray_normal)
	if hit == null:
		return
	aim_point = hit
	var look_target := Vector3(aim_point.x, body.global_position.y, aim_point.z)
	if body.global_position.distance_squared_to(look_target) > 0.01:
		body.look_at(look_target, Vector3.UP)


func _respawn() -> void:
	_release_grapple()
	_end_slide()
	global_position = _spawn_position
	velocity = Vector3.ZERO


func _on_died() -> void:
	# Placeholder until restart/checkpoint behaviour is decided (GDD 13.3).
	get_tree().reload_current_scene.call_deferred()

## so the player sees more of the screen in the direction they're shooting.
@export_range(0.0, 0.5) var aim_lead := 0.15
## Largest shake offset in metres, and how fast a shake dies down (per second).
@export var max_shake := 0.6
@export var shake_decay := 2.5

@onready var _camera: Camera3D = find_child("Camera3D") as Camera3D

var _trauma := 0.0


func _ready() -> void:
	add_to_group("camera_rig")
	if target:
		global_position = target.global_position


func _physics_process(delta: float) -> void:
	if target == null:
		return
	var goal := target.global_position
	var player := target as Player
	if player and aim_lead > 0.0:
		# Sideways only: the aim point sits at gun height, which shouldn't lift the camera.
		var lead := player.aim_point - goal
		lead.y = 0.0
		goal += lead * aim_lead
	global_position = global_position.lerp(goal, 1.0 - exp(-follow_sharpness * delta))


func _process(delta: float) -> void:
	if _camera == null:
		return
	_trauma = maxf(_trauma - shake_decay * delta, 0.0)
	var amount := _trauma * _trauma * max_shake
	_camera.h_offset = randf_range(-amount, amount)
	_camera.v_offset = randf_range(-amount, amount)


## strength 0-1; shakes add up to a maximum of 1.
func shake(strength: float) -> void:
	_trauma = minf(_trauma + strength, 1.0)

class_name Combat
## Damage rules shared by every gun, the player's and the enemies': the Infinite Pistol's instant kill,
## poison, explosions and chain lightning.

## Player shots hit walls and enemies; enemy shots hit walls and the player (see the layer names in Project Settings).
const PLAYER_SHOT_MASK := 1 | 4
const ENEMY_SHOT_MASK := 1 | 2


## The Health of a body that was hit, or null for walls and props.
static func health_of(node: Node) -> Health:
	if node == null:
		return null
	return node.get_node_or_null("Health") as Health


## One projectile or beam hitting target.
static func hit(target: Node, weapon: WeaponData, amount: int, kind := Health.Kind.NORMAL) -> void:
	var health := health_of(target)
	if health == null or health.is_dead():
		return
	if weapon.instant_kill_normal:
		var enemy := target as Enemy
		if enemy and enemy.tier == Enemy.Tier.NORMAL:
			enemy.killed_by_infinite_pistol = true
			health.take_damage(health.current, Health.Kind.INSTAKILL)
			return
		amount = weapon.elite_damage
	health.take_damage(amount, kind)
	if weapon.poison_dps > 0:
		health.apply_poison(weapon.poison_dps, weapon.poison_duration)


## Damages everything on the other team within radius of center.
static func explode(context: Node, center: Vector3, radius: float, amount: int, color: Color, hurts_enemies := true) -> void:
	Effects.flash(context, center, color, radius, 0.35)
	Effects.shake(context, 0.12)
	var group := "enemies" if hurts_enemies else "player"
	for node in context.get_tree().get_nodes_in_group(group):
		var body := node as Node3D
		if body == null:
			continue
		var offset := body.global_position - center
		offset.y = 0.0
		if offset.length() <= radius + 0.5:
			var health := health_of(body)
			if health:
				health.take_damage(amount, Health.Kind.EXPLOSION)


## Lightning from the enemy that was hit to up to weapon.chain_count others nearby, one after another.
static func chain(context: Node, first: Node3D, weapon: WeaponData, amount: int) -> void:
	var already_hit: Array[Node3D] = [first]
	var from := first
	for i in weapon.chain_count:
		var next := nearest_enemy(context, from.global_position, weapon.chain_range, already_hit)
		if next == null:
			return
		Effects.lightning(context, from.global_position + Vector3.UP, next.global_position + Vector3.UP, weapon.projectile_color)
		var health := health_of(next)
		if health:
			health.take_damage(maxi(1, roundi(amount * weapon.chain_damage_factor)), Health.Kind.CHAIN)
		already_hit.append(next)
		from = next


## Closest living enemy to point on the ground plane, within max_distance. Null if none.
static func nearest_enemy(context: Node, point: Vector3, max_distance: float, exclude: Array[Node3D] = []) -> Node3D:
	var best: Node3D = null
	var best_distance := max_distance
	for node in context.get_tree().get_nodes_in_group("enemies"):
		var enemy := node as Node3D
		if enemy == null or enemy in exclude:
			continue
		var offset := enemy.global_position - point
		offset.y = 0.0
		if offset.length() < best_distance:
			best_distance = offset.length()
			best = enemy
	return best

class_name Effects
## Small throwaway visuals made in code: flashes, lightning, beams and floating damage numbers.
## Everything here frees itself when its tween ends.

static var _materials := {}


## Unshaded glowing material, cached per colour so hundreds of bullets share a few materials.
static func glow_material(color: Color, transparent := false) -> StandardMaterial3D:
	var key := "%s|%s" % [color, transparent]
	if not _materials.has(key):
		var material := StandardMaterial3D.new()
		material.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
		material.albedo_color = color
		if transparent:
			material.transparency = BaseMaterial3D.TRANSPARENCY_ALPHA
			material.blend_mode = BaseMaterial3D.BLEND_MODE_ADD
			material.no_depth_test = false
		material.disable_receive_shadows = true
		_materials[key] = material
	return _materials[key]


## A sphere that grows and fades: muzzle flashes, explosions, spawn and teleport puffs.
static func flash(context: Node, position: Vector3, color: Color, radius := 1.0, duration := 0.3) -> void:
	var root := _root(context)
	if root == null:
		return
	var mesh := SphereMesh.new()
	mesh.radius = 1.0
	mesh.height = 2.0
	mesh.radial_segments = 16
	mesh.rings = 8
	var material := glow_material(color, true).duplicate() as StandardMaterial3D
	mesh.material = material
	var instance := MeshInstance3D.new()
	instance.mesh = mesh
	instance.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	root.add_child(instance)
	instance.global_position = position
	instance.scale = Vector3.ONE * radius * 0.3
	var tween := instance.create_tween().set_parallel()
	tween.tween_property(instance, "scale", Vector3.ONE * radius, duration).set_ease(Tween.EASE_OUT).set_trans(Tween.TRANS_CUBIC)
	tween.tween_property(material, "albedo_color:a", 0.0, duration)
	tween.chain().tween_callback(instance.queue_free)


## A straight glowing line from a to b (railgun beam, lightning arcs).
static func beam(context: Node, a: Vector3, b: Vector3, color: Color, width := 0.12, duration := 0.25) -> void:
	var root := _root(context)
	var length := a.distance_to(b)
	if root == null or length < 0.01:
		return
	var mesh := BoxMesh.new()
	mesh.size = Vector3(width, width, length)
	var material := glow_material(color, true).duplicate() as StandardMaterial3D
	mesh.material = material
	var instance := MeshInstance3D.new()
	instance.mesh = mesh
	instance.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	root.add_child(instance)
	instance.global_position = (a + b) * 0.5
	var up := Vector3.UP if absf((b - a).normalized().y) < 0.99 else Vector3.RIGHT
	instance.look_at(b, up)
	var tween := instance.create_tween().set_parallel()
	tween.tween_property(material, "albedo_color:a", 0.0, duration)
	tween.tween_property(instance, "scale", Vector3(0.2, 0.2, 1.0), duration)
	tween.chain().tween_callback(instance.queue_free)


## Jagged lightning between two points, built from a few short beams.
static func lightning(context: Node, a: Vector3, b: Vector3, color: Color, duration := 0.22) -> void:
	var points := [a]
	var segments := 5
	for i in range(1, segments):
		var t := float(i) / segments
		var jitter := Vector3(randf_range(-0.35, 0.35), randf_range(-0.2, 0.2), randf_range(-0.35, 0.35))
		points.append(a.lerp(b, t) + jitter)
	points.append(b)
	for i in points.size() - 1:
		beam(context, points[i], points[i + 1], color, 0.07, duration)


## Floating text that rises and fades, e.g. damage numbers.
static func float_text(context: Node, position: Vector3, text: String, color: Color, size := 1.0) -> void:
	var root := _root(context)
	if root == null:
		return
	var label := Label3D.new()
	label.text = text
	label.modulate = color
	label.outline_modulate = Color(0, 0, 0, 0.85)
	label.font_size = int(56 * size)
	label.outline_size = int(14 * size)
	label.pixel_size = 0.01
	label.billboard = BaseMaterial3D.BILLBOARD_ENABLED
	label.no_depth_test = true
	label.render_priority = 10
	root.add_child(label)
	label.global_position = position + Vector3(randf_range(-0.3, 0.3), 0, randf_range(-0.3, 0.3))
	var tween := label.create_tween().set_parallel()
	tween.tween_property(label, "global_position:y", label.global_position.y + 1.4, 0.8).set_ease(Tween.EASE_OUT).set_trans(Tween.TRANS_QUAD)
	tween.tween_property(label, "modulate:a", 0.0, 0.35).set_delay(0.45)
	tween.chain().tween_callback(label.queue_free)


## Shakes every camera rig in the scene.
static func shake(context: Node, strength := 0.3) -> void:
	if context.is_inside_tree():
		context.get_tree().call_group("camera_rig", "shake", strength)

		return null
	var tree := context.get_tree()
	return tree.current_scene if tree.current_scene else tree.root

class_name Health
extends Node
## Reusable hit points. Add as a child of the player or any enemy and listen to its signals.
## Also runs poison damage-over-time and can pop up floating damage numbers.

signal changed(current: int, maximum: int)
signal damaged(amount: int, kind: Kind)
signal died

## What caused the damage, so damage numbers can be coloured.
enum Kind { NORMAL, POISON, EXPLOSION, CHAIN, INSTAKILL }

const KIND_COLORS := {
	Kind.NORMAL: Color(1, 1, 1),
	Kind.POISON: Color(0.45, 1, 0.3),
	Kind.EXPLOSION: Color(1, 0.6, 0.2),
	Kind.CHAIN: Color(0.45, 0.85, 1),
	Kind.INSTAKILL: Color(0.5, 1, 1),
}
const POISON_TICK := 0.5

@export var max_health := 100
## Floating numbers above the owner when it takes damage (on for enemies).
@export var show_damage_numbers := false
## Height of the damage numbers above the owner's origin.
@export var number_height := 1.8

var current := 0
## Set by the owner, e.g. during a dash's invincibility frames.
var invulnerable := false
## Debug cheat: ignore all damage.
var infinite_health := false

var _poison_dps := 0
var _poison_time_left := 0.0
var _poison_tick_left := 0.0


func _ready() -> void:
	current = max_health
	set_process(false)


func is_dead() -> bool:
	return current <= 0


func is_poisoned() -> bool:
	return _poison_time_left > 0.0


func take_damage(amount: int, kind := Kind.NORMAL) -> void:
	if invulnerable or infinite_health or is_dead() or amount <= 0:
		return
	current = maxi(current - amount, 0)
	changed.emit(current, max_health)
	damaged.emit(amount, kind)
	if show_damage_numbers:
		_show_number(amount, kind)
	if current == 0:
		_poison_time_left = 0.0
		died.emit()


func heal(amount: int) -> void:
	if is_dead() or amount <= 0:
		return
	current = mini(current + amount, max_health)
	changed.emit(current, max_health)


## Fills health back up without any effects (training dummy, cheats).
func reset() -> void:
	current = max_health
	changed.emit(current, max_health)


## Poison deals dps every second for duration. A new dose refreshes the timer and keeps the stronger dps.
func apply_poison(dps: int, duration: float) -> void:
	if is_dead() or dps <= 0 or duration <= 0.0:
		return
	_poison_dps = maxi(_poison_dps, dps) if is_poisoned() else dps
	_poison_time_left = maxf(_poison_time_left, duration)
	if not is_processing():
		_poison_tick_left = POISON_TICK
	set_process(true)


func _process(delta: float) -> void:
	_poison_time_left -= delta
	_poison_tick_left -= delta
	if _poison_tick_left <= 0.0:
		_poison_tick_left += POISON_TICK
		take_damage(maxi(1, roundi(_poison_dps * POISON_TICK)), Kind.POISON)
	if _poison_time_left <= 0.0 or is_dead():
		_poison_time_left = 0.0
		set_process(false)


func _show_number(amount: int, kind: Kind) -> void:
	var body := get_parent() as Node3D
	if body == null:
		return
	var text := "INSTAKILL!" if kind == Kind.INSTAKILL else str(amount)
	var size := 1.5 if kind == Kind.INSTAKILL else (0.8 if kind == Kind.POISON else 1.0)
	Effects.float_text(self, body.global_position + Vector3.UP * number_height, text, KIND_COLORS[kind], size)

## What the Infinite Pistol makes a normal enemy spill: 2-3 weapons (jackpot_weapons first, random
## ones for the rest), ammo, a pile of cash and some health.
static func jackpot(enemy: Enemy) -> void:
	var origin := enemy.global_position
	var weapons: Array[WeaponData] = enemy.jackpot_weapons.duplicate()
	var count := maxi(weapons.size(), randi_range(2, 3))
	while weapons.size() < count:
		weapons.append(WeaponLibrary.random_drop())
	var start_angle := randf() * TAU
	for i in weapons.size():
		var angle := start_angle + TAU * i / weapons.size()
		spawn_weapon(enemy, WeaponSlot.new(weapons[i]), origin + Vector3.UP, landing_spot(enemy, origin, 2.6, 3.4, angle))
	for i in 2:
		spawn_item(enemy, LootPickup.Kind.AMMO, 0, origin + Vector3.UP, landing_spot(enemy, origin, 1.2, 2.2))
	for i in 4:
		spawn_item(enemy, LootPickup.Kind.MONEY, randi_range(8, 15), origin + Vector3.UP, landing_spot(enemy, origin, 1.0, 2.5))
	spawn_item(enemy, LootPickup.Kind.HEALTH, 25, origin + Vector3.UP, landing_spot(enemy, origin, 1.0, 2.0))
	Effects.flash(enemy, origin + Vector3.UP, Rarity.color(Rarity.Tier.INFINITE), 3.0, 0.5)
	Effects.shake(enemy, 0.4)
	enemy.get_tree().call_group("hud", "show_message", "JACKPOT!", 1.5)


## A normal kill: some cash, maybe ammo, health or a weapon (chances set on the enemy).
static func normal_drop(enemy: Enemy) -> void:
	var origin := enemy.global_position
	var cash := randi_range(enemy.money_drop.x, enemy.money_drop.y)
	if cash > 0:
		spawn_item(enemy, LootPickup.Kind.MONEY, cash, origin + Vector3.UP, landing_spot(enemy, origin, 0.5, 1.5))
	if randf() < enemy.ammo_drop_chance:
		spawn_item(enemy, LootPickup.Kind.AMMO, 0, origin + Vector3.UP, landing_spot(enemy, origin, 0.5, 1.5))
	if randf() < enemy.health_drop_chance:
		spawn_item(enemy, LootPickup.Kind.HEALTH, 15, origin + Vector3.UP, landing_spot(enemy, origin, 0.5, 1.5))
	if randf() < enemy.weapon_drop_chance:
		spawn_weapon(enemy, WeaponSlot.new(WeaponLibrary.random_drop()), origin + Vector3.UP, landing_spot(enemy, origin, 1.5, 2.5))


## A point on the floor near origin, min to max metres away (at angle, or a random one),
## pulled back so it doesn't end up inside or behind a wall.
static func landing_spot(context: Node3D, origin: Vector3, min_distance: float, max_distance: float, angle := NAN) -> Vector3:
	var space := context.get_world_3d().direct_space_state
	if is_nan(angle):
		angle = randf() * TAU
	var start := origin + Vector3.UP * 0.5
	var direction := Vector3(cos(angle), 0.0, sin(angle))
	var target := start + direction * randf_range(min_distance, max_distance)
	var wall := space.intersect_ray(PhysicsRayQueryParameters3D.create(start, target, 1))
	if not wall.is_empty():
		target = start + direction * maxf(start.distance_to(wall.position) - 0.7, 0.0)
	var floor_hit := space.intersect_ray(PhysicsRayQueryParameters3D.create(target + Vector3.UP * 2.0, target + Vector3.DOWN * 10.0, 1))
	target.y = floor_hit.position.y if not floor_hit.is_empty() else origin.y
	return target


static func _root(context: Node) -> Node:
	var tree := context.get_tree()
	return tree.current_scene if tree.current_scene else tree.root

## Fades to color, swaps the scene, then fades back in. Ignored while a change is already running.
func change_scene(path: String, color := Color.BLACK, fade_time := 0.45) -> void:

		return
	_busy = true
	_rect.color = Color(color, 0.0)
	var fade_out := create_tween()
	fade_out.tween_property(_rect, "color:a", 1.0, fade_time)
	await fade_out.finished
	get_tree().paused = false
	get_tree().change_scene_to_file(path)
	# Wait for the new scene to be in place before revealing it.
	await get_tree().process_frame
	await get_tree().process_frame
	var fade_in := create_tween()
	fade_in.tween_property(_rect, "color:a", 0.0, fade_time)
	await fade_in.finished
	_busy = false


## A quick full-screen flash without changing scene.
func flash(color: Color, duration := 0.4) -> void:
	_rect.color = Color(color, 0.8)
	create_tween().tween_property(_rect, "color:a", 0.0, duration)

class_name DialogueBox
extends CanvasLayer
## Dialogue panel along the bottom of the screen, with typewriter text.
## Call start() with a speaker and lines. E (talk) shows the whole line, then moves to the next one.
## Emits finished after the last line is closed.
## ask() shows one line with answer buttons (click, or press 1/2/3) and emits choice_made with the index.

signal started
signal finished
signal choice_made(index: int)

## Characters revealed per second.
@export var reveal_speed := 45.0

@onready var _speaker_label: Label = %SpeakerLabel
@onready var _text_label: Label = %TextLabel
@onready var _continue_hint: Label = %ContinueHint
@onready var _choice_row: HBoxContainer = %ChoiceRow

var _lines := PackedStringArray()
var _index := 0
var _revealed := 0.0
## Answers for the current question; empty for normal lines.
var _choices := PackedStringArray()


func _ready() -> void:
	add_to_group("dialogue_box")
	visible = false
	set_process(false)


func is_open() -> bool:
	return visible


func start(speaker: String, lines: PackedStringArray) -> void:
	if lines.is_empty():
		push_warning("DialogueBox.start() called with no lines.")
		return
	_speaker_label.text = speaker
	_lines = lines
	_choices = PackedStringArray()
	_choice_row.visible = false
	visible = true
	set_process(true)
	_show_line(0)
	started.emit()


## A question with answers. Await choice_made for the picked index.
func ask(speaker: String, question: String, options: PackedStringArray) -> void:
	start(speaker, PackedStringArray([question]))
	_choices = options


func close() -> void:
	visible = false
	set_process(false)
	_choice_row.visible = false
	finished.emit()


func _process(delta: float) -> void:
	if not _is_line_done():
		_revealed += reveal_speed * delta
		_text_label.visible_characters = int(_revealed)
	var waiting_for_answer := not _choices.is_empty()
	_continue_hint.visible = _is_line_done() and not waiting_for_answer
	if waiting_for_answer and _is_line_done() and not _choice_row.visible:
		_show_choices()


func _unhandled_input(event: InputEvent) -> void:
	if not visible:
		return
	if not _choices.is_empty() and _choice_row.visible:
		var key := event as InputEventKey
		if key and key.pressed and not key.echo:
			var index := key.physical_keycode - KEY_1
			if index >= 0 and index < _choices.size():
				get_viewport().set_input_as_handled()
				_pick(index)
		return
	if not event.is_action_pressed("talk"):
		return
	get_viewport().set_input_as_handled()
	if not _is_line_done():
		_text_label.visible_characters = -1
	elif not _choices.is_empty():
		return
	elif _index + 1 < _lines.size():
		_show_line(_index + 1)
	else:
		close()


func _show_line(index: int) -> void:
	_index = index
	_text_label.text = _lines[index]
	_text_label.visible_characters = 0
	_revealed = 0.0
	_continue_hint.visible = false
	_continue_hint.text = "[E] continue" if index + 1 < _lines.size() else "[E] close"


func _show_choices() -> void:
	for child in _choice_row.get_children():
		child.queue_free()
	for i in _choices.size():
		var button := Button.new()
		button.text = "[%d]  %s" % [i + 1, _choices[i]]
		button.add_theme_font_size_override("font_size", 28)
		button.custom_minimum_size = Vector2(220, 56)
		button.pressed.connect(_pick.bind(i))
		_choice_row.add_child(button)
	_choice_row.visible = true
	(_choice_row.get_child(0) as Button).grab_focus.call_deferred()


func _pick(index: int) -> void:
	_choices = PackedStringArray()
	close()
	choice_made.emit(index)


func _is_line_done() -> bool:
	return _text_label.visible_characters < 0 or _text_label.visible_characters >= _text_label.get_total_character_count()

extends CanvasLayer
## Gameplay HUD: the health/dash bar (the team's sprite: three red health segments and a green dash block that
## refills like water, see shaders/), currencies, current objective and short pop-up messages,
## plus the weapon panel, the item card for weapons on the floor, a boss health bar and a red flash when hurt.
## Top right: the Infinite Pistol's charge coins (PistolCharge).

## The style meter gets added here later.

## Gap between a pickup's position on screen and its item card.
const CARD_OFFSET := Vector2(70, -40)
const SCREEN_MARGIN := 16.0

@export var player: Player

@onready var _root: Control = $Root
@onready var _health_segments: Array[TextureRect] = [%HealthSegment1, %HealthSegment2, %HealthSegment3]
@onready var _dash_fill: TextureRect = %DashFill
@onready var _currency_label: Label = %CurrencyLabel
@onready var _objective_label: Label = %ObjectiveLabel
@onready var _message_label: Label = %MessageLabel

var _card_target: WeaponPickup
var _boss: Enemy
var _boss_box: VBoxContainer
var _boss_name: Label
var _boss_bar: ProgressBar
var _hurt_flash: ColorRect
var _health := 0
var _max_health := 1
## Health shown pale in the segments: lags behind real health after a hit, then drains down to it.
var _ghost_health := 0.0
var _ghost_tween: Tween
var _dash_level := 1.0


func _ready() -> void:
	add_to_group("hud")
	_message_label.modulate.a = 0.0
	_build_weapon_ui()
	_build_boss_bar()
	_build_hurt_flash()

	if player == null:
		push_warning("HUD has no player assigned.")
		return
	if not player.is_node_ready():
		await player.ready
	player.health.changed.connect(_on_health_changed)
	player.health.damaged.connect(_on_player_damaged)

	_pistol_charge.watch(player.arsenal)

func _process(_delta: float) -> void:
	if player:
		_set_dash_level(player.dash_recharge_progress())
	_place_card()
	if _boss_box.visible and not is_instance_valid(_boss):
		_boss_box.visible = false

## Shows text in the middle of the screen for a moment, then fades it out.
func show_message(text: String, duration := 2.5) -> void:

		_message_tween.kill()
	_message_label.modulate.a = 1.0

## Shows a health bar for this enemy at the top of the screen until it dies.
func track_boss(enemy: Enemy) -> void:

	_boss_bar.value = enemy.health.current

func _refresh_weapons() -> void:

	_pistol_charge.refresh(player.arsenal)
	if is_instance_valid(_card_target):
		_on_focus_changed(_card_target)


func _on_focus_changed(pickup: WeaponPickup) -> void:
	_card_target = pickup
	if pickup == null or pickup.slot == null:
		_card.visible = false
		return
	_card.show_weapon(pickup.slot, player.arsenal.current(), player.arsenal.slot_to_replace())
	_place_card()


## Keeps the item card beside its pickup on screen, inside the screen edges.
func _place_card() -> void:
	if not _card.visible:
		return
	if not is_instance_valid(_card_target) or _card_target.slot == null:
		_card.visible = false
		return
	var camera := get_viewport().get_camera_3d()
	if camera == null:
		return
	var screen := camera.unproject_position(_card_target.global_position + Vector3.UP)
	var view := _root.size
	var position := screen + CARD_OFFSET
	if position.x + _card.size.x > view.x - SCREEN_MARGIN:
		position.x = screen.x - CARD_OFFSET.x - _card.size.x
	position.x = clampf(position.x, SCREEN_MARGIN, view.x - _card.size.x - SCREEN_MARGIN)
	position.y = clampf(position.y - _card.size.y * 0.5, SCREEN_MARGIN, view.y - _card.size.y - SCREEN_MARGIN)
	_card.position = position


func _on_health_changed(current: int, maximum: int) -> void:
	_health = current
	_max_health = maxi(maximum, 1)
	if _ghost_tween:
		_ghost_tween.kill()
	if current >= _ghost_health:
		_ghost_health = current
	else:
		_ghost_tween = create_tween()
		_ghost_tween.tween_interval(0.35)
		_ghost_tween.tween_method(_set_ghost_health, _ghost_health, float(current), 0.3)
	_update_health_segments()


func _set_ghost_health(value: float) -> void:
	_ghost_health = value
	_update_health_segments()


## Each segment holds a third of max health and drains from its right edge, the last segment first.
func _update_health_segments() -> void:
	var per_segment := float(_max_health) / _health_segments.size()
	for i in _health_segments.size():
		var material := _health_segments[i].material as ShaderMaterial
		material.set_shader_parameter("fill", clampf((_health - i * per_segment) / per_segment, 0.0, 1.0))
		material.set_shader_parameter("ghost", clampf((_ghost_health - i * per_segment) / per_segment, 0.0, 1.0))


## The dash block fills like water while recharging and flashes when the dash is ready again.
func _set_dash_level(level: float) -> void:
	if is_equal_approx(level, _dash_level):
		return
	if level >= 1.0 and _dash_level < 1.0:
		_dash_fill.modulate = Color(2.2, 2.2, 2.2)
		create_tween().tween_property(_dash_fill, "modulate", Color.WHITE, 0.3)
	_dash_level = level
	(_dash_fill.material as ShaderMaterial).set_shader_parameter("level", level)

	_hurt_flash.color.a = 0.35

class_name PistolCharge
extends Control
## Top-right HUD row for the Infinite Pistol: one coin per kill it needs (kills_per_charge, 7: the gambler's
## lucky number). Coins are cyan while empty. Each kill with another gun fills the next coin with gold, which
## then hops (the team's Aseprite animation). When the last coin fills, every coin spins like a slot machine
## reel and they stop one by one, left to right, on gold: the pistol is loaded. Firing it empties the row.
## Art: assets/textures/ui/pistol_charge/. Shader: shaders/slot_coin.gdshader. Built by the HUD.

const GOLD := preload("res://assets/textures/ui/pistol_charge/coin_gold.png")
const GOLD_HOP := preload("res://assets/textures/ui/pistol_charge/coin_gold_hop.png")
const EMPTY := preload("res://assets/textures/ui/pistol_charge/coin_empty.png")
const SHADER := preload("res://shaders/slot_coin.gdshader")

## Screen pixels per art pixel.
const PIXEL_SCALE := 4
## Art pixels from one coin to the next (16 px coin, 1 px gap), and the coin sprite's size.
const COIN_STEP := 17
const COIN_SIZE := Vector2(16, 18)
## The Aseprite animation: gold coins rest for 1 s, then hop up a pixel for 0.1 s.
const REST_TIME := 1.0
const HOP_TIME := 0.1
## Seconds for gold to fill a coin, and between coins when several fill at once.
const FILL_TIME := 0.3
const FILL_STAGGER := 0.08
## Slot machine: the first reel spins SPIN_TURNS coins in SPIN_TIME seconds; each reel after it spins
## SPIN_TURNS_STEP more and SPIN_TIME_STEP longer, so they stop left to right. Turns must stay even (gold).
const SPIN_TURNS := 12
const SPIN_TURNS_STEP := 2
const SPIN_TIME := 0.9
const SPIN_TIME_STEP := 0.14
## How fast (coins per second) a reel must spin for full motion blur.
const FULL_BLUR_SPEED := 30.0

var _coins: Array[TextureRect] = []
## Per coin: its gold level (0 to 1), and its reel position now and last frame.
var _fill: Array[float] = []
var _spin: Array[float] = []
var _last_spin: Array[float] = []
## Coins shown (or filling to) gold.
var _filled := 0
var _shown_once := false
## Set by Arsenal.pistol_charged: the next refresh plays the slot machine.
var _jackpot_pending := false
var _spinning := false
var _idle_time := 0.0
## Running fill and spin animations, stopped when the row resets.
var _tweens: Array[Tween] = []


func _ready() -> void:
	mouse_filter = MOUSE_FILTER_IGNORE
	visible = false


## Hooks the row to the player's weapons. Called once by the HUD.
func watch(arsenal: Arsenal) -> void:
	arsenal.pistol_charged.connect(func() -> void: _jackpot_pending = true)


func refresh(arsenal: Arsenal) -> void:
	visible = arsenal.pistol != null
	if not visible:
		_shown_once = false
		return
	var count := arsenal.pistol.data.kills_per_charge
	if count != _coins.size():
		_build(count)
	var loaded := not arsenal.pistol.is_out_of_ammo()
	var target := count if loaded else mini(arsenal.pistol_kills, count)
	if not _shown_once:
		# First time on screen (level loaded): show the current charge without animating.
		_shown_once = true
		_jackpot_pending = false
		_filled = target
		for i in count:
			_set_fill(1.0 if i < target else 0.0, i)
		return
	if target < _filled:
		# Fired (or reset): every coin goes back to cyan.
		for tween in _tweens:
			tween.kill()
		_tweens.clear()
		_stop_spin()
		for i in count:
			_set_fill(1.0 if i < target else 0.0, i)
		_filled = target
	var delay := 0.0
	for i in range(_filled, target):
		var tween := _new_tween()
		tween.tween_interval(delay)
		tween.tween_method(_set_fill.bind(i), 0.0, 1.0, FILL_TIME)
		tween.tween_callback(_hop.bind(i))
		delay += FILL_STAGGER
	_filled = target
	if loaded and _jackpot_pending:
		_jackpot_pending = false
		_new_tween().tween_interval(delay + FILL_TIME + 0.1).finished.connect(_play_slot_machine)


func _process(delta: float) -> void:
	if not visible:
		return
	# Idle hop from the Aseprite file, for every full gold coin.
	_idle_time = fmod(_idle_time + delta, REST_TIME + HOP_TIME)
	var hopping := _idle_time >= REST_TIME and not _spinning
	for i in _coins.size():
		var material := _coins[i].material as ShaderMaterial
		if _fill[i] >= 1.0 and not _coins[i].has_meta("hop"):
			material.set_shader_parameter("gold_tex", GOLD_HOP if hopping else GOLD)
		# Motion blur follows how fast each reel turns.
		var speed := absf(_spin[i] - _last_spin[i]) / maxf(delta, 0.001)
		_last_spin[i] = _spin[i]
		material.set_shader_parameter("blur", clampf(speed / FULL_BLUR_SPEED, 0.0, 1.0))


func _build(count: int) -> void:
	for coin in _coins:
		coin.queue_free()
	_coins.clear()
	_fill.clear()
	_spin.clear()
	_last_spin.clear()
	var coin_size := COIN_SIZE * PIXEL_SCALE
	for i in count:
		var coin := TextureRect.new()
		coin.texture = GOLD
		coin.texture_filter = CanvasItem.TEXTURE_FILTER_NEAREST
		coin.stretch_mode = TextureRect.STRETCH_SCALE
		coin.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
		coin.mouse_filter = MOUSE_FILTER_IGNORE
		coin.position = Vector2(i * COIN_STEP * PIXEL_SCALE, 0)
		coin.size = coin_size
		var material := ShaderMaterial.new()
		material.shader = SHADER
		material.set_shader_parameter("gold_tex", GOLD)
		material.set_shader_parameter("empty_tex", EMPTY)
		coin.material = material
		add_child(coin)
		_coins.append(coin)
		_fill.append(0.0)
		_spin.append(0.0)
		_last_spin.append(0.0)
	custom_minimum_size = Vector2(((count - 1) * COIN_STEP + COIN_SIZE.x) * PIXEL_SCALE, coin_size.y)


func _set_fill(value: float, index: int) -> void:
	_fill[index] = value
	(_coins[index].material as ShaderMaterial).set_shader_parameter("fill", value)


## A coin that just turned gold hops once, like the Aseprite's second frame.
func _hop(index: int) -> void:
	var material := _coins[index].material as ShaderMaterial
	_coins[index].set_meta("hop", true)
	material.set_shader_parameter("gold_tex", GOLD_HOP)
	await get_tree().create_timer(HOP_TIME).timeout
	if is_instance_valid(_coins[index]):
		_coins[index].remove_meta("hop")
		material.set_shader_parameter("gold_tex", GOLD)


## All coins spin like slot machine reels and stop left to right on gold, each with a small bounce and a flash.
func _play_slot_machine() -> void:
	if _filled < _coins.size():
		return
	_spinning = true
	for i in _coins.size():
		var material := _coins[i].material as ShaderMaterial
		var turns := float(SPIN_TURNS + SPIN_TURNS_STEP * i)
		_spin[i] = 0.0
		_last_spin[i] = 0.0
		material.set_shader_parameter("reel", 1.0)
		var tween := _new_tween()
		tween.tween_method(_set_spin.bind(i), 0.0, turns + 0.15, SPIN_TIME + SPIN_TIME_STEP * i) \
				.set_trans(Tween.TRANS_QUART).set_ease(Tween.EASE_OUT)
		tween.tween_method(_set_spin.bind(i), turns + 0.15, turns, 0.12).set_trans(Tween.TRANS_SINE)
		tween.tween_callback(material.set_shader_parameter.bind("flash", 1.0))
		tween.tween_method(func(v: float) -> void: material.set_shader_parameter("flash", v), 1.0, 0.0, 0.3)
		tween.parallel().tween_method(func(v: float) -> void: material.set_shader_parameter("reel", v), 1.0, 0.0, 0.15)
		if i == _coins.size() - 1:
			tween.tween_callback(_stop_spin)


func _new_tween() -> Tween:
	for i in range(_tweens.size() - 1, -1, -1):
		if not _tweens[i].is_valid():
			_tweens.remove_at(i)
	var tween := create_tween()
	_tweens.append(tween)
	return tween


func _set_spin(value: float, index: int) -> void:
	_spin[index] = value
	(_coins[index].material as ShaderMaterial).set_shader_parameter("spin", value)


func _stop_spin() -> void:
	_spinning = false
	for i in _coins.size():
		var material := _coins[i].material as ShaderMaterial
		_spin[i] = 0.0
		_last_spin[i] = 0.0
		material.set_shader_parameter("spin", 0.0)
		material.set_shader_parameter("reel", 0.0)
		material.set_shader_parameter("flash", 0.0)

		_select(wrapi(_selected - 1, 0, _rows.size()))
	elif event.is_action_pressed("ui_accept"):
		chosen.emit(_selected)
	else:
		return
	get_viewport().set_input_as_handled()

	row.root.add_child(row.bar)

	row.label.offset_right = -RIGHT_MARGIN
	row.root.add_child(row.label)
	return row

	var click := event as InputEventMouseButton
	if click and click.pressed and click.button_index == MOUSE_BUTTON_LEFT:
		accept_event()

	_selected = index
	_refresh()

	for i in _rows.size():
		var selected := i == _selected
		_rows[i].bar.visible = selected
		_rows[i].label.add_theme_color_override("font_color", DARK if selected else CYAN)

		tags.append("Automatic")

	if weapon.homing_turn_rate > 0.0:
		tags.append("Homing")
	if weapon.max_fire_rate > weapon.fire_rate:

class_name Arsenal
extends Node
## The player's weapons: SLOT_COUNT slots for guns found on the floor, plus the Infinite Pistol once the
## Orange Trooper hands it over. Handles shooting, ammo, swapping, picking up and dropping weapons, and
## charging the Infinite Pistol with kills.
## Saved to GameState.loadout on every change, so it survives scene changes and reloads.
## Expects to be a child of the Player.

## Slots, ammo, the active weapon or the pistol's charge changed.
signal changed
## The weapon on the floor that E would pick up changed. Null when there is none.
signal focus_changed(pickup: WeaponPickup)
## The Infinite Pistol just filled up from kills (or the cheat). The HUD plays its slot machine.
signal pistol_charged

const SLOT_COUNT := 2
## Seconds between repeated "out of ammo" messages.
const WARNING_COOLDOWN := 1.5

## The gun model is parented here (it turns with the player's aim).
@export var hand: Node3D

var slots: Array[WeaponSlot] = []
## The Infinite Pistol, or null before the Orange Trooper hands it over.
var pistol: WeaponSlot
## Kills since the pistol's last charge.
var pistol_kills := 0
## Index into loadout(): the slots first, then the pistol.
var active := 0
## The nearest weapon on the floor in reach, or null.
var focused_pickup: WeaponPickup
## The gun currently in the hand, or null.
var held: Weapon

var _nearby: Array[WeaponPickup] = []
var _shot_queued := false
var _warning_cooldown := 0.0

@onready var _player := get_parent() as Player


func _ready() -> void:
	GameState.enemy_killed.connect(_on_enemy_killed)
	changed.connect(_save)
	_load(GameState.loadout)
	_equip_active()


## Everything the player can switch between, in order: slot 1, slot 2, then the Infinite Pistol.
func loadout() -> Array[WeaponSlot]:
	var list: Array[WeaponSlot] = slots.duplicate()
	if pistol:
		list.append(pistol)
	return list


func current() -> WeaponSlot:
	var list := loadout()
	return list[active] if active >= 0 and active < list.size() else null


## Called by the Player every physics frame.
func update(aim_point: Vector3, input_enabled: bool, delta: float) -> void:
	_warning_cooldown = maxf(_warning_cooldown - delta, 0.0)
	_update_focus()
	var slot := current()
	var holding := input_enabled and Input.is_action_pressed("shoot")
	if held:
		held.trigger_held = holding and slot != null and not slot.is_out_of_ammo()
	# Semi-automatic guns fire once per click (queued from the input event so fast clicks aren't lost);
	# automatic guns fire for as long as the button is held.
	var wants_shot := _shot_queued or (holding and slot != null and slot.data.automatic)
	_shot_queued = false
	if not input_enabled or not wants_shot:
		return
	if slot == null:
		_warn("No weapon! Find one first.")
		return
	if slot.is_out_of_ammo():
		_warn_empty(slot)
		return
	var on_miss := _refund_pistol if slot == pistol and slot.data.refund_on_miss else Callable()
	if held.fire(aim_point, on_miss):
		slot.ammo -= slot.data.ammo_per_shot
		changed.emit()


func _unhandled_input(event: InputEvent) -> void:
	if _player == null or not _player.input_enabled:
		return
	if event.is_action_pressed("shoot"):
		_shot_queued = true
	elif event.is_action_pressed("interact") and focused_pickup:
		get_viewport().set_input_as_handled()
		pick_up(focused_pickup)
	elif event.is_action_pressed("swap_weapon"):
		var mouse := event as InputEventMouseButton
		cycle(-1 if mouse and mouse.button_index == MOUSE_BUTTON_WHEEL_UP else 1)
	elif event.is_action_pressed("weapon_1"):
		select(0)
	elif event.is_action_pressed("weapon_2"):
		select(1)
	elif event.is_action_pressed("weapon_3"):
		select(2)


func select(index: int) -> void:
	if index == active or index < 0 or index >= loadout().size():
		return
	active = index
	_equip_active()
	changed.emit()


func cycle(step: int) -> void:
	var count := loadout().size()
	if count < 2:
		return
	active = posmod(active + step, count)
	_equip_active()
	changed.emit()


func pick_up(pickup: WeaponPickup) -> void:
	if not is_instance_valid(pickup) or not pickup.can_be_picked():
		return
	var slot := pickup.take()
	AudioManager.play_sfx("weapon_pickup")
	add_weapon(slot)
	_message("Picked up %s" % slot.data.display_name, 1.2)


## Puts a weapon in a slot and equips it: a free slot first, then one that's out of ammo, then the
## active one. The weapon it replaces is dropped on the floor (or thrown away if it was empty).
func add_weapon(slot: WeaponSlot) -> void:
	var index := _slot_for_new_weapon()
	if index < slots.size():
		_drop(slots[index])
		slots[index] = slot
	else:
		slots.append(slot)
		index = slots.size() - 1
	active = index
	_equip_active()
	changed.emit()


func give_infinite_pistol(charged := true) -> void:
	pistol = WeaponSlot.new(WeaponLibrary.INFINITE_PISTOL, 1 if charged else 0)
	pistol_kills = 0
	active = loadout().size() - 1
	_equip_active()
	changed.emit()


func charge_pistol() -> void:
	if pistol:
		pistol.ammo = pistol.data.max_ammo
		pistol_kills = 0
		pistol_charged.emit()
		changed.emit()


## Adds fraction of each slot weapon's max ammo (not the pistol's). Returns false if all were full.
func refill_ammo(fraction: float) -> bool:
	var refilled := false
	for slot in slots:
		refilled = slot.refill(fraction) or refilled
	if refilled:
		changed.emit()
	return refilled


func clear() -> void:
	slots.clear()
	pistol = null
	pistol_kills = 0
	active = 0
	_equip_active()
	changed.emit()


## The weapon a pickup would swap out right now, or null if there's a free slot.
func slot_to_replace() -> WeaponSlot:
	var index := _slot_for_new_weapon()
	return slots[index] if index < slots.size() else null


## Kills still needed for the next pistol shot (0 when it's loaded).
func pistol_kills_needed() -> int:
	if pistol == null or not pistol.is_out_of_ammo():
		return 0
	return pistol.data.kills_per_charge - pistol_kills


func add_nearby(pickup: WeaponPickup) -> void:
	if not pickup in _nearby:
		_nearby.append(pickup)


func remove_nearby(pickup: WeaponPickup) -> void:
	_nearby.erase(pickup)


func _update_focus() -> void:
	var best: WeaponPickup = null
	var best_distance := INF
	for pickup in _nearby.duplicate():
		if not is_instance_valid(pickup):
			_nearby.erase(pickup)
			continue
		if not pickup.can_be_picked():
			continue
		var distance: float = pickup.global_position.distance_squared_to(_player.global_position)
		if distance < best_distance:
			best_distance = distance
			best = pickup
	if best == focused_pickup:
		return
	if is_instance_valid(focused_pickup):
		focused_pickup.set_focused(false)
	focused_pickup = best
	if best:
		best.set_focused(true)
	focus_changed.emit(best)


func _slot_for_new_weapon() -> int:
	if slots.size() < SLOT_COUNT:
		return slots.size()
	for i in slots.size():
		if slots[i].is_out_of_ammo():
			return i
	return active if active < slots.size() else 0


func _drop(slot: WeaponSlot) -> void:
	var origin := _player.global_position
	if slot.is_out_of_ammo():
		Effects.float_text(_player, origin + Vector3.UP * 2.2, "Tossed the empty %s" % slot.data.display_name, Color(0.8, 0.8, 0.8), 0.6)
		return
	Loot.spawn_weapon(_player, slot, origin + Vector3.UP, Loot.landing_spot(_player, origin, 1.6, 2.2))


func _equip_active() -> void:
	if held:
		held.queue_free()
		held = null
	var slot := current()
	if slot == null or hand == null:
		return
	held = Weapon.new()
	hand.add_child(held)
	held.setup(slot.data)


func _on_enemy_killed(enemy: Node) -> void:
	if pistol == null or pistol.is_full():
		return
	# Kills made by the pistol itself don't recharge it.
	if enemy is Enemy and (enemy as Enemy).killed_by_infinite_pistol:
		return
	pistol_kills += 1
	if pistol_kills >= pistol.data.kills_per_charge:
		pistol_kills = 0
		pistol.ammo = pistol.data.max_ammo
		_message("INFINITE PISTOL CHARGED!  [%s]" % _pistol_key(), 2.0)
		pistol_charged.emit()
	changed.emit()


func _refund_pistol() -> void:
	if pistol == null or pistol.is_full():
		return
	pistol.ammo += 1
	changed.emit()
	Effects.float_text(_player, _player.global_position + Vector3.UP * 2.2, "Missed! The bullet returns to the chamber.", Color(0.6, 1, 1), 0.6)


func _warn_empty(slot: WeaponSlot) -> void:
	if slot == pistol:
		_warn("Infinite Pistol is empty: %d more kill(s) with other guns to recharge." % pistol_kills_needed())
	else:
		_warn("%s is out of ammo! Grab another weapon [E] or swap [Q]." % slot.data.display_name)


func _warn(text: String) -> void:
	if _warning_cooldown > 0.0:
		return
	_warning_cooldown = WARNING_COOLDOWN
	_message(text, 1.5)


func _message(text: String, duration: float) -> void:
	get_tree().call_group("hud", "show_message", text, duration)


func _pistol_key() -> String:
	return str(loadout().find(pistol) + 1)


func _save() -> void:
	var saved_slots := []
	for slot in slots:
		saved_slots.append(slot.to_dict())
	GameState.loadout = {
		"slots": saved_slots,
		"pistol": pistol.ammo if pistol else -1,
		"pistol_kills": pistol_kills,
		"active": active,
	}


func _load(saved: Dictionary) -> void:
	slots.clear()
	for entry in saved.get("slots", []):
		var slot := WeaponSlot.from_dict(entry)
		if slot and slots.size() < SLOT_COUNT:
			slots.append(slot)
	var pistol_ammo: int = saved.get("pistol", -1)
	pistol = WeaponSlot.new(WeaponLibrary.INFINITE_PISTOL, pistol_ammo) if pistol_ammo >= 0 else null
	pistol_kills = saved.get("pistol_kills", 0)
	active = clampi(saved.get("active", 0), 0, maxi(loadout().size() - 1, 0))

class_name Projectile
extends Node3D
## One bullet, orb or shard. Each physics frame it ray-casts along the step it is about to move, so fast
## bullets can't skip through thin walls (and it works the same with Jolt or Godot physics).
## Its behaviour - damage, pierce, ricochet, explosion, poison, chain lightning, homing - comes from a WeaponData.
## Made in code by Weapon.fire(); no scene needed.

## How far homing projectiles look for a target.
const HOMING_RANGE := 14.0

var weapon: WeaponData
var collision_mask := Combat.PLAYER_SHOT_MASK
## Height the projectile dips to after leaving the muzzle (at 45 degrees), then holds. NAN keeps the muzzle's
## height. Lets a gun held at chest height fire bullets low enough for low walls to block.
var cruise_height := NAN

var _velocity := Vector3.ZERO
var _age := 0.0
var _pierce_left := 0
var _bounces_left := 0
var _hit_an_enemy := false
var _exclude: Array[RID] = []
var _on_miss := Callable()


## Call right after adding the projectile to the scene. on_miss runs if it ends without hitting anyone.
func launch(data: WeaponData, origin: Vector3, direction: Vector3, mask: int, on_miss := Callable()) -> void:
	weapon = data
	collision_mask = mask
	_on_miss = on_miss
	_pierce_left = data.pierce
	_bounces_left = data.bounces
	global_position = origin
	# Keeps any downward angle (the player shooting from the grapple); enemies pass flat directions.
	_velocity = direction.normalized() * data.projectile_speed
	_build_mesh()
	_face_velocity()


## Instant beam (railgun): damages every enemy on the line up to the first wall.
## Returns true if it hit at least one enemy.
static func fire_beam(context: Node3D, data: WeaponData, origin: Vector3, direction: Vector3, mask: int) -> bool:
	var space := context.get_world_3d().direct_space_state
	var end := origin + direction * data.projectile_range
	var exclude: Array[RID] = []
	var hit_enemy := false
	var pierce_left := data.pierce
	for i in 32:
		var hit := space.intersect_ray(PhysicsRayQueryParameters3D.create(origin, end, mask, exclude))
		if hit.is_empty():
			break
		var collider := hit.collider as Node
		if Combat.health_of(collider) == null:
			end = hit.position
			break
		hit_enemy = true
		Combat.hit(collider, data, data.damage)
		Effects.flash(context, hit.position, data.projectile_color, 0.6, 0.15)
		exclude.append(hit.rid)
		if pierce_left == 0:
			end = hit.position
			break
		pierce_left -= 1
	Effects.beam(context, origin, end, data.projectile_color * 1.5, 0.22, 0.35)
	Effects.beam(context, origin, end, Color(1, 1, 1), 0.07, 0.2)
	return hit_enemy


func _physics_process(delta: float) -> void:
	if weapon.homing_turn_rate > 0.0:
		_steer(delta)
	var from := global_position
	var to := from + _velocity * delta
	if not is_nan(cruise_height):
		to.y = move_toward(from.y, cruise_height, weapon.projectile_speed * delta)
	var query := PhysicsRayQueryParameters3D.create(from, to, collision_mask, _exclude)
	var hit := get_world_3d().direct_space_state.intersect_ray(query)
	if hit.is_empty():
		global_position = to
	elif not _on_hit(hit):
		_finish()
		return
	_age += delta
	if _age >= weapon.lifetime():
		_finish()


## Returns true if the projectile keeps flying (pierce or ricochet).
func _on_hit(hit: Dictionary) -> bool:
	var collider := hit.collider as Node
	var point: Vector3 = hit.position
	var exploding := weapon.explosion_radius > 0.0
	var hurts_enemies := collision_mask & 4 != 0
	if Combat.health_of(collider):
		_hit_an_enemy = true
		Combat.hit(collider, weapon, weapon.damage)
		if exploding:
			Combat.explode(self, point, weapon.explosion_radius, weapon.explosion_damage, weapon.projectile_color, hurts_enemies)
		if weapon.chain_count > 0 and collider is Node3D:
			Combat.chain(self, collider, weapon, weapon.damage)
		Effects.flash(self, point, weapon.projectile_color, 0.4, 0.12)
		if _pierce_left == 0:
			return false
		_pierce_left -= 1
		_exclude.append(hit.rid)
		global_position = point
		return true

	# Hit a wall or prop.
	var normal: Vector3 = hit.normal
	normal.y = 0.0
	if exploding:
		Combat.explode(self, point + normal * 0.2, weapon.explosion_radius, weapon.explosion_damage, weapon.projectile_color, hurts_enemies)
	if _bounces_left > 0 and normal.length_squared() > 0.01:
		_bounces_left -= 1
		normal = normal.normalized()
		_velocity = _velocity.bounce(normal)
		_velocity.y = 0.0
		global_position = point + normal * 0.05
		_face_velocity()
		return true
	if not exploding:
		Effects.flash(self, point, weapon.projectile_color, 0.25, 0.1)
	return false


func _finish() -> void:
	if not _hit_an_enemy and _on_miss.is_valid():
		_on_miss.call()
	queue_free()


func _steer(delta: float) -> void:
	var target := Combat.nearest_enemy(self, global_position, HOMING_RANGE)
	if target == null:
		return
	var desired := target.global_position - global_position
	desired.y = 0.0
	var angle := _velocity.signed_angle_to(desired, Vector3.UP)
	var turn := clampf(angle, -weapon.homing_turn_rate * delta, weapon.homing_turn_rate * delta)
	_velocity = _velocity.rotated(Vector3.UP, turn)
	_face_velocity()


func _face_velocity() -> void:
	if _velocity.length_squared() > 0.001:
		# Straight down from the grapple, UP can't be the up vector.
		var up := Vector3.UP if absf(_velocity.normalized().y) < 0.99 else Vector3.FORWARD
		look_at(global_position + _velocity, up)


func _build_mesh() -> void:
	var sphere := SphereMesh.new()
	sphere.radius = weapon.projectile_size
	sphere.height = weapon.projectile_size * 2.0
	sphere.radial_segments = 10
	sphere.rings = 5
	# Brighter than 1.0 so the environment's glow picks it up.
	sphere.material = Effects.glow_material(weapon.projectile_color * 2.2)
	var mesh_instance := MeshInstance3D.new()
	mesh_instance.mesh = sphere
	mesh_instance.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
	# Fast bullets are stretched along their path so they read as tracers.
	mesh_instance.scale = Vector3(1.0, 1.0, clampf(weapon.projectile_speed / 16.0, 1.0, 4.0))
	add_child(mesh_instance)

## Weapon rarity tiers, colour-coded like Borderlands. The rank shown on item cards is the tier number + 1.

enum Tier { COMMON, UNCOMMON, RARE, EPIC, LEGENDARY, INFINITE }

const NAMES := ["Common", "Uncommon", "Rare", "Epic", "Legendary", "Infinite"]
const COLORS := [
	Color(0.92, 0.92, 0.92),
	Color(0.3, 0.95, 0.35),

class_name Weapon
extends Node3D
## A gun in someone's hand, built from a WeaponData with setup(). fire() shoots toward a point.
## Ammo is not tracked here: the player's Arsenal (or an enemy) decides whether a shot is allowed.

signal fired

## Closer than this, the target is beside or behind the muzzle, so fire straight ahead instead.
const MIN_AIM_DISTANCE := 1.5

var data: WeaponData
## Set every physics frame by whoever holds the gun. Drives spin-up weapons.
var trigger_held := false
var collision_mask := Combat.PLAYER_SHOT_MASK

var _model: Node3D
var _muzzle_local := Vector3.ZERO
var _cooldown := 0.0
var _spin := 0.0
var _kick := 0.0



func setup(weapon: WeaponData) -> void:
	data = weapon
	_model = weapon.build_model()
	add_child(_model)
	_muzzle_local = weapon.muzzle_in(_model)


func muzzle_position() -> Vector3:
	return global_transform * _muzzle_local


func current_fire_rate() -> float:
	if data.max_fire_rate <= data.fire_rate:
		return data.fire_rate
	return lerpf(data.fire_rate, data.max_fire_rate, _spin)


## 0 to 1, how far a spin-up weapon has wound up.
func spin() -> float:
	return _spin


func _physics_process(delta: float) -> void:
	if data == null:
		return
	_cooldown = maxf(_cooldown - delta, 0.0)
	if data.spin_up_time > 0.0:
		var direction := 1.0 if trigger_held else -2.0
		_spin = clampf(_spin + direction * delta / data.spin_up_time, 0.0, 1.0)
	# Recoil: the model jumps back a little on each shot and slides forward again.
	_kick = move_toward(_kick, 0.0, delta * 1.2)
	_model.position.z = _kick


func can_fire() -> bool:
	return data != null and _cooldown <= 0.0


## Shoots toward target, angled down if it's below the muzzle (shooting from the grapple). Returns false
## while on cooldown. on_miss runs if the shot hits no enemy (the Infinite Pistol refunds its bullet).
func fire(target: Vector3, on_miss := Callable()) -> bool:
	if not can_fire():
		return false
	var origin := muzzle_position()
	var direction := target - origin
	if direction.length() < MIN_AIM_DISTANCE:
		direction = -global_basis.z
		direction.y = 0.0
	direction = direction.normalized()

	if data.hitscan:
		if not Projectile.fire_beam(self, data, origin, direction, collision_mask) and on_miss.is_valid():
			on_miss.call()
	else:
		for i in data.pellets:
			var projectile := Projectile.new()
			get_tree().current_scene.add_child(projectile)
			var miss_callback := on_miss if i == 0 else Callable()
			projectile.launch(data, origin, direction.rotated(Vector3.UP, _spread_angle(i)), collision_mask, miss_callback)

	_cooldown = 1.0 / current_fire_rate()
	_kick = 0.06 + 0.02 * data.pellets
	Effects.flash(self, origin, data.projectile_color, 0.22, 0.07)
	fired.emit()
	return true


func _spread_angle(pellet: int) -> float:
	var spread := deg_to_rad(data.spread_degrees)
	if data.pellets <= 1:
		return randf_range(-0.5, 0.5) * spread
	# Shotguns: pellets fanned evenly across the cone, with a little jitter.
	return lerpf(-0.5 * spread, 0.5 * spread, float(pellet) / (data.pellets - 1)) + randf_range(-0.03, 0.03)

func muzzle_in(model_root: Node3D) -> Vector3:

		return muzzle_position

	return Vector3(bounds.get_center().x, bounds.position.y + bounds.size.y * 0.72, bounds.position.z)


## Bounding box of every mesh under root, in root's space. Works before root is in the tree.
static func mesh_bounds(root: Node3D) -> AABB:
	var result := AABB()
	var first := true
	for node in root.find_children("*", "MeshInstance3D", true, false):
		var mesh_instance := node as MeshInstance3D
		var to_root := Transform3D.IDENTITY
		var walker: Node = mesh_instance
		while walker != root and walker is Node3D:
			to_root = (walker as Node3D).transform * to_root
			walker = walker.get_parent()
		var box := to_root * mesh_instance.get_aabb()
		result = box if first else result.merge(box)
		first = false
	return result

	preload("res://resources/weapons/toy_gun.tres"),
	INFINITE_PISTOL,

static func find(id: StringName) -> WeaponData:
	for weapon in ALL:
		if weapon.id == id:
			return weapon
	return null


## Everything except the Infinite Pistol, sorted from common to legendary.
static func droppable() -> Array[WeaponData]:
	var result: Array[WeaponData] = []
	for weapon in ALL:
		if weapon.rarity != Rarity.Tier.INFINITE:
			result.append(weapon)
	result.sort_custom(func(a: WeaponData, b: WeaponData) -> bool: return a.rarity < b.rarity)
	return result


## A random weapon, picked by rarity weight (Rarity.DROP_WEIGHTS), at least min_tier.
static func random_drop(min_tier := Rarity.Tier.COMMON) -> WeaponData:
	var pool := droppable()
	var total := 0.0
	for weapon in pool:
		if weapon.rarity >= min_tier:
			total += _weight(weapon, pool)
	var roll := randf() * total
	for weapon in pool:
		if weapon.rarity < min_tier:
			continue
		roll -= _weight(weapon, pool)
		if roll <= 0.0:
			return weapon
	return pool.back()


## A tier's weight is shared between the weapons in it, so adding guns doesn't change tier odds.
static func _weight(weapon: WeaponData, pool: Array[WeaponData]) -> float:
	var same_tier := pool.filter(func(w: WeaponData) -> bool: return w.rarity == weapon.rarity).size()
	return Rarity.DROP_WEIGHTS[weapon.rarity] / maxf(same_tier, 1.0)

	data = weapon
	if weapon:
		ammo = weapon.max_ammo if ammo_left < 0 else mini(ammo_left, weapon.max_ammo)

	return ammo < data.ammo_per_shot

	if is_full():
		return false
	ammo = mini(data.max_ammo, ammo + maxi(1, ceili(data.max_ammo * fraction)))
	return true

	return {"id": String(data.id), "ammo": ammo}

	var weapon := WeaponLibrary.find(StringName(d.get("id", "")))
	if weapon == null:
		return null
	return WeaponSlot.new(weapon, int(d.get("ammo", -1)))

// One red segment of the HUD health bar. Drains from its right edge as fill drops, with the cut slanted
// like the segment's sides. Health just lost stays pale (ghost) until the HUD catches it up to fill.
shader_type canvas_item;

uniform float fill : hint_range(0.0, 1.0) = 1.0;
uniform float ghost : hint_range(0.0, 1.0) = 1.0;
// How far the bottom edge sits right of the top edge, as a fraction of the texture's width.
uniform float slant = 0.0;
uniform vec4 ghost_color : source_color = vec4(1.0, 0.86, 0.8, 1.0);

void fragment() {
	// Snap to the sprite's pixels so the cut stays crisp pixel art.
	vec2 uv = (floor(UV / TEXTURE_PIXEL_SIZE) + 0.5) * TEXTURE_PIXEL_SIZE;
	float along = (uv.x - slant * uv.y) / max(1.0 - slant, 0.001);
	vec4 tex = texture(TEXTURE, UV);
	// Exact when full or empty: the sides aren't perfectly parallel, so the slanted cut alone would clip a corner.
	float shown = fill >= 0.999 ? 2.0 : (fill <= 0.001 ? -2.0 : fill);
	float pale = ghost >= 0.999 ? 2.0 : (ghost <= 0.001 ? -2.0 : ghost);
	if (along > max(shown, pale)) {
		COLOR.a = 0.0;
	} else if (along > shown) {
		COLOR = vec4(ghost_color.rgb, tex.a * COLOR.a);
	}
}

// One coin of the Infinite Pistol charge row (top right of the HUD, see PistolCharge).
//   fill: gold rises from the bottom like liquid when a kill lands (0 = cyan coin, 1 = gold coin).
//   reel/spin: when all coins are gold the coin becomes a slot machine reel. The reel is a drum of gold and
//   cyan coins taking turns; spin scrolls it down (in coins), reel bends and shades it like a real drum, and
//   blur smears it while it spins fast. An even whole spin lands on a gold coin.
//   flash: whitens the coin, used when a reel stops.
shader_type canvas_item;

uniform sampler2D gold_tex : filter_nearest;
uniform sampler2D empty_tex : filter_nearest;
uniform float fill : hint_range(0.0, 1.0) = 0.0;
uniform float spin = 0.0;
uniform float reel : hint_range(0.0, 1.0) = 0.0;
uniform float blur : hint_range(0.0, 1.0) = 0.0;
uniform float flash : hint_range(0.0, 1.0) = 0.0;
uniform vec4 liquid_top : source_color = vec4(1.0, 0.99, 0.72, 1.0);

// The reel at height y (in coins): even coins are gold, odd coins are cyan.
vec4 reel_coin(float x, float y) {
	vec2 p = vec2(x, fract(y));
	return mod(floor(y), 2.0) < 0.5 ? texture(gold_tex, p) : texture(empty_tex, p);
}

void fragment() {
	vec2 px = TEXTURE_PIXEL_SIZE;
	// Snap to the sprite's pixels so it stays crisp pixel art.
	vec2 uv = (floor(UV / px) + 0.5) * px;
	vec4 col;
	if (reel > 0.0) {
		// -1 at the top of the window, 1 at the bottom. The drum squashes coins near the edges.
		float yc = UV.y * 2.0 - 1.0;
		float bent = asin(clamp(yc * 0.85, -1.0, 1.0)) / asin(0.85);
		float y = mix(yc, bent, reel) * 0.5 + 0.5 - spin;
		col = vec4(0.0);
		for (int i = 0; i < 5; i++) {
			col += reel_coin(uv.x, y + (float(i) / 4.0 - 0.5) * blur * 0.7);
		}
		col /= 5.0;
		col.rgb *= 1.0 - 0.55 * reel * yc * yc;
	} else {
		vec4 gold = texture(gold_tex, uv);
		col = texture(empty_tex, uv);
		if (fill >= 1.0) {
			col = gold;
		} else if (fill > 0.0) {
			// The coin covers sprite rows 1 to 17; the wavy gold surface rises through them.
			float wave = 0.6 * sin(uv.x * 18.0 + TIME * 9.0) + 0.4 * sin(uv.x * 31.0 - TIME * 13.0);
			float surface = (mix(17.0, 1.0, fill) + wave * 1.2) * px.y;
			if (uv.y > surface && gold.a > 0.0) {
				col = gold;
				bool outline = dot(gold.rgb, vec3(1.0)) < 0.2;
				if (!outline && uv.y < surface + px.y * 1.5) {
					col.rgb = liquid_top.rgb;
				}
			}
		}
	}
	col.rgb = mix(col.rgb, vec3(1.0), flash);
	COLOR = col;
}

    vec3 diff = (color - decimated) * d;
    vec3 ditherPattern = vec3(ditherBayer(xy));
    return decimate(color + (step(ditherPattern, diff) / d), d);
}
float ditherBayer(const in float val, const in vec2 xy) { return ditherBayer(vec3(val), xy, DITHER_BAYER_PRECISION).r; }

float get_star(in vec2 uv){

	float star_flare = texture(stars, uv).r;
	float star_cell = texture(star_cells, uv).r;
	star_flare = star_flare * clamp(sin((TIME/10.+star_cell*PI*2.))*100.-99., star_cell*.005, 1);

	return star_flare;
}

void fragment() {
	// Called for every pixel the material is visible on.
	// 1. Planning + 2. Cloud up
	// A cellular noise can make some convincing cloud shapes.
	vec3 cloud = texture(clouds, UV).rgb;
	
	// 3. Dither, 4. Layers
	// figure out the edges, by first doing a bit of posterization
	vec3 edges = decimate(cloud+edge_size, steps);
	cloud = decimate(cloud, steps);
	// mask out an area to dither
	vec3 dither_area = texture(dither_areas, UV).rgb;
	float dither_value = ditherBayer(dither_area.r, FRAGCOORD.xy);
	// multiply the dither area with the edges,
	float should_dither = clamp(sign(edges.r-cloud.r),0,1);
	should_dither *= dither_value;
	
	// 5. Rift!
	vec3 rift = texture(rifts, UV).rgb;
	float rift_mask = decimate(rift, 2).r;
	float rift_edge = step(.5, rift_mask) - rift_mask;
	rift_mask = step(.6,rift_mask);
	cloud.r = max(0, cloud.r-rift_mask);
	cloud.r = mix(cloud.r, cloud.r * 3., rift_edge);
	// Apply the dithering to our cloud effect!
	cloud = mix(cloud, vec3(cloud.r+1./steps), should_dither);
	float star = get_star(UV);
	cloud = mix(texture(color_ramp, vec2(cloud.r, 0)).rgb, vec3(1.), smoothstep(.5,1., star));
	COLOR.rgb = cloud.rgb;
}
